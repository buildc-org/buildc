### Base Image
FROM ${BASE}

## Module layers

# ---- buildc-org/hello ----
RUN --mount=type=bind,from=ctx,source=/modules/buildc-org/hello,target=/ctx/modules/buildc-org/hello \
    --mount=type=bind,from=ctx,source=/lib,target=/ctx/lib \
    --mount=type=bind,from=ctx,source=/generated/example.d/buildc-org/hello.sh,target=/ctx/module.sh \
    --mount=type=bind,from=buildc,source=/buildc,target=/ctx/buildc \
    --mount=type=cache,target=/var/cache \
    --mount=type=cache,target=/var/log \
    --mount=type=tmpfs,target=/tmp \
    bash /ctx/module.sh

# ---- flavour gate ----
ARG FLAVOUR

# ---- image version ----
ARG IMAGE_VERSION=dev

# ---- base family ----
ARG FAMILY

# ---- image identity ----
ARG IMAGE_ID="example"
ARG IMAGE_NAME="Example"
ARG IMAGE_REGISTRY=""

# ---- os-release ----
RUN --mount=type=bind,from=buildc,source=/buildc,target=/ctx/buildc \
    IMAGE_VERSION="${IMAGE_VERSION}" IMAGE_ID="${IMAGE_ID}" IMAGE_NAME="${IMAGE_NAME}" IMAGE_REGISTRY="${IMAGE_REGISTRY}" /ctx/buildc os-release
