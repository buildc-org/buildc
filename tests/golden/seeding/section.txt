### Base Image
FROM quay.io/fedora/fedora-bootc:44

## Module layers

# ---- hello ----
RUN --mount=type=bind,from=ctx,source=/modules/hello,target=/ctx/modules/hello \
    --mount=type=bind,from=ctx,source=/lib,target=/ctx/lib \
    --mount=type=bind,from=ctx,source=/generated/example.d/hello.sh,target=/ctx/module.sh \
    --mount=type=bind,from=buildc,source=/buildc,target=/ctx/buildc \
    --mount=type=cache,target=/var/cache \
    --mount=type=cache,target=/var/log \
    --mount=type=tmpfs,target=/tmp \
    bash /ctx/module.sh
